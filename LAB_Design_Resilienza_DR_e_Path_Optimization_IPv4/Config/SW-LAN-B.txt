! Configurazione SW-LAN-B
! Ruolo: Switch L2 Accesso Sito B (Disaster Recovery)
!
hostname SW-LAN-B
!
no ip domain-lookup
line con 0
 exec-timeout 0 0
 logging synchronous
!
vlan 10
 name SITO_B_LAN
!
interface Ethernet0/0
 description TO_EDGE_B
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast edge
 no shutdown
!
interface Ethernet0/1
 description TO_PC_B
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast edge
 no shutdown
!
end
