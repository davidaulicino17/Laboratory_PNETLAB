! Configurazione Base R1 (Border/Core)
enable
configure terminal

! ---------------------------------------------------------
! 1. Hostname e Disabilitazione DNS
! ---------------------------------------------------------
hostname R1
no ip domain-lookup

! ---------------------------------------------------------
! 2. Configurazione Enable Mode
! ---------------------------------------------------------
enable secret cisco

! ---------------------------------------------------------
! 3. Configurazione Interfacce
! ---------------------------------------------------------

! Loopback
interface Loopback0
 description Router_ID_Loopback
 ip address 1.1.1.1 255.255.255.255
 exit

! Verso R3 (OSPF)
interface Ethernet0/0
 description Link_Verso_R3_OSPF
 ip address 10.1.13.1 255.255.255.0
 no shutdown
 exit

! Verso R4 (EIGRP)
interface Ethernet0/1
 description Link_Verso_R4_EIGRP
 ip address 10.1.14.1 255.255.255.0
 no shutdown
 exit

! Verso R5 (ISP BGP) - Subnet /30
interface Ethernet0/2
 description Link_Verso_R5_ISP
 ip address 203.0.113.1 255.255.255.252
 no shutdown
 exit

! Verso R2 (Backdoor)
interface Ethernet0/3
 description Link_Verso_R2_Backdoor
 ip address 10.1.12.1 255.255.255.0
 no shutdown
 exit

end
write memory
