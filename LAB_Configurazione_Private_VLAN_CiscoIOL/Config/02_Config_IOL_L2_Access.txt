! Switch Access (Immagine IOL L2)
enable
configure terminal
hostname Access_L2

! 1. Impostazione VTP
vtp mode transparent

! 2. Creazione VLAN Secondarie
vlan 101
 private-vlan isolated
 name PVLAN_ISOL
 exit

vlan 102
 private-vlan community
 name PVLAN_COMM
 exit

! 3. Creazione VLAN Primaria e Associazione
vlan 100
 private-vlan primary
 private-vlan association 101-102
 name PVLAN_PRIMA
 exit

! 4. Configurazione Porta Promiscuous (Verso Core L3)
interface Ethernet0/0
 description UPLINK_TO_CORE
 switchport trunk encapsulation dot1q
 switchport mode private-vlan promiscuous
 switchport private-vlan mapping 100 101-102
 no shutdown
 exit

! 5. Configurazione Porte Host - ISOLATED (PC1, PC2)
interface range Ethernet0/1 - 2
 description HOST_ISOLATED
 switchport mode private-vlan host
 switchport private-vlan host-association 100 101
 spanning-tree portfast
 no shutdown
 exit

! 6. Configurazione Porte Host - COMMUNITY (PC3, PC4)
interface range Ethernet0/3 - 4
 description HOST_COMMUNITY
 switchport mode private-vlan host
 switchport private-vlan host-association 100 102
 spanning-tree portfast
 no shutdown
 exit

end
write memory
